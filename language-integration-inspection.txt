import java.util.Locale;
import java.util.Objects;

public final class Translator {
  private static Locale locale;

  private Translator() {
  }

  public static void setLocale(String languageCode) {
    if (languageCode == null || languageCode.isBlank()) {
      throw new IllegalArgumentException("The client language code must not be empty");
    }
    String[] parts = languageCode.replace('-', '_').split("_", 3);
    if (!parts[0].matches("[A-Za-z]{2,8}")
        || (parts.length > 1 && !parts[1].matches("[A-Za-z0-9]{2,8}"))) {
      throw new IllegalArgumentException("Unsupported client language code: " + languageCode);
    }
    String country = parts.length > 1 ? parts[1] : "";
    String variant = parts.length > 2 ? parts[2] : "";
    locale = new Locale(parts[0], country, variant);
  }

  public static String toLowercaseWithLocale(String value) {
    if (locale == null) {
      throw new IllegalStateException("The JS client language has not been supplied to JEI");
    }
    return Objects.requireNonNull(value, "value").toLowerCase(locale);
  }
}

  private static String jsonEscape(String value) {
    StringBuilder escaped = new StringBuilder(value.length());

      char character = value.charAt(index);
      switch (character) {
        case '"':
          escaped.append("\\\"");
          break;
        case '\\':
          escaped.append("\\\\");
          break;
        case '\b':
          escaped.append("\\b");
          break;

          break;
        case '\r':
          escaped.append("\\r");
          break;
        case '\t':
          escaped.append("\\t");
          break;

  private static List<String> decodeTooltipData(String encoded) {
    List<String> lines = new ArrayList<>();
    int offset = 0;
    while (offset < encoded.length()) {
      int separator = encoded.indexOf(':', offset);
      if (separator < 0 || separator == offset) {
        throw new IllegalStateException("Malformed client tooltip data length");
      }
      int lineLength;
      try {
        lineLength = Integer.parseInt(encoded.substring(offset, separator));
      } catch (NumberFormatException exception) {
        throw new IllegalStateException("Malformed client tooltip data length", exception);
      }
      int lineStart = separator + 1;
      if (lineLength < 0 || lineLength > encoded.length() - lineStart) {
        throw new IllegalStateException("Client tooltip data length is out of bounds");
      }
      lines.add(encoded.substring(lineStart, lineStart + lineLength));
      offset = lineStart + lineLength;
    }
    return List.copyOf(lines);
  }
